`timescale 1ns / 1ps 
module Full_adder( 
    input a, 
    input b, 
    input ci, 
    output sum, 
    output carry 
); 

    wire w1, w2, w3, w4, w5; 

    // Sum = a ^ b ^ ci 
    xor (w1, a, b); 
    xor (sum, w1, ci); 

    // Carry = (a & b) | (b & ci) | (a & ci) 
    and (w2, a, b); 
    and (w3, b, ci); 
    and (w4, a, ci); 
    or  (w5, w2, w3); 
    or  (carry, w5, w4); 
endmodule
