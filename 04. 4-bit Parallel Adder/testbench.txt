    wire Cout;        // Carry-out

        .A(A),
        .B(B),
        .Cin(Cin),
        .Sum(Sum),
        .Cout(Cout)
    );
    initial begin

        A = 4'b0000; B = 4'b0000; Cin = 0;
        #10;

        // Test case 1: 4'b0001 + 4'b0001 with Cin = 0

        // Test case 2: 4'b1111 + 4'b1111 with Cin = 0
        A = 4'b1111; B = 4'b1111; Cin = 0;
        #10;
        $display("A = %b, B = %b, Cin = %b, Sum = %b, Cout = %b", A, B, Cin, Sum, Cout);

        // Test case 3: 4'b0101 + 4'b0011 with Cin = 1
        A = 4'b0101; B = 4'b0011; Cin = 1;
        #10;
        $display("A = %b, B = %b, Cin = %b, Sum = %b, Cout = %b", A, B, Cin, Sum, Cout);

        // Test case 4: 4'b1001 + 4'b1001 with Cin = 1

        // End simulation
